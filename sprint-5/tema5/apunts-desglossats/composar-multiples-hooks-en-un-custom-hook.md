**Com es poden compondre múltiples hooks en un Custom Hook?**

Un Custom Hook pot cridar qualsevol hook de React (`useState`, `useEffect`, `useCallback`, `useMemo`) **i** qualsevol altre Custom Hook. La composició és el mecanisme que permet construir lògica complexa a partir de peces simples.

| // La cadena de composició:// Hook Base → Hook de Domini → Hook de Feature → ComponentuseWindowSize()        // base: una sola responsabilitat  ↓useResponsiveLayout()  // domini: composa useWindowSize \+ useMediaQuery  ↓useDashboard()         // feature: composa layout \+ fetch \+ permisos  ↓Dashboard component    // únicament JSX |
| :---- |

**Patró 1: Composar Hooks Base en Hooks de domini**

| // HOOKS BASE (atòmics, una responsabilitat cadascun):function useWindowSize() {  const \[mida, setMida\] \= useState({    amplada: window.innerWidth,    alçada: window.innerHeight,  });  useEffect(() \=\> {    const fn \= () \=\> setMida({ amplada: window.innerWidth, alçada: window.innerHeight });    window.addEventListener('resize', fn);    return () \=\> window.removeEventListener('resize', fn);  }, \[\]);  return mida;}function useMediaQuery(query: string): boolean {  const \[coincideix, setCoincideix\] \= useState(    () \=\> window.matchMedia(query).matches  );  useEffect(() \=\> {    const media \= window.matchMedia(query);    const fn \= (e: MediaQueryListEvent) \=\> setCoincideix(e.matches);    media.addEventListener('change', fn);    return () \=\> media.removeEventListener('change', fn);  }, \[query\]);  return coincideix;}// HOOK DE DOMINI: composa els dos hooks base en una abstracció de nivell superiorfunction useResponsiveLayout() {  const { amplada } \= useWindowSize();         // ← hook base 1  const esMobil \= useMediaQuery('(max-width: 768px)');   // ← hook base 2  const esTablet \= useMediaQuery('(max-width: 1024px)'); // ← hook base 2 (altra query)  const prefereixFosc \= useMediaQuery('(prefers-color-scheme: dark)'); // ← hook base 2  return {    amplada,    esMobil,    esTablet,    esDesktop: \!esTablet,    prefereixFosc,    columnes: esMobil ? 1 : esTablet ? 2 : 3,    // ↑ lògica de negoci derivada de la composició  };}// Ús al component: API d'alt nivell sense saber quins hooks hi ha darrerefunction Dashboard() {  const { esMobil, columnes } \= useResponsiveLayout();  return (    \<div style={{ gridTemplateColumns: \`repeat(${columnes}, 1fr)\` }}\>      {esMobil ? \<MobileNav /\> : \<DesktopNav /\>}    \</div\>  );} |

**Patró 2: Composar Fetch \+ estat local \+ lògica de negoci**

| // HOOK BASE 1: fetch genèricfunction useFetch\<T\>(url: string | null) {  const \[dades, setDades\] \= useState\<T | null\>(null);  const \[loading, setLoading\] \= useState(true);  const \[error, setError\] \= useState\<Error | null\>(null);  useEffect(() \=\> {    if (\!url) return;    let ignore \= false;    setLoading(true);    fetch(url)      .then(r \=\> { if (\!r.ok) throw new Error(\`HTTP ${r.status}\`); return r.json(); })      .then(d \=\> { if (\!ignore) { setDades(d); setLoading(false); } })      .catch(e \=\> { if (\!ignore) { setError(e); setLoading(false); } });    return () \=\> { ignore \= true; };  }, \[url\]);  return { dades, loading, error };}// HOOK BASE 2: debounce genèricfunction useDebounce\<T\>(valor: T, retard: number): T {  const \[valorRetardat, setValorRetardat\] \= useState(valor);  useEffect(() \=\> {    const timer \= setTimeout(() \=\> setValorRetardat(valor), retard);    return () \=\> clearTimeout(timer);  }, \[valor, retard\]);  return valorRetardat;}// HOOK BASE 3: localStorage genèricfunction useLocalStorage\<T\>(clau: string, inicial: T) {  const \[valor, setValor\] \= useState\<T\>(() \=\> {    try { return JSON.parse(localStorage.getItem(clau) ?? 'null') ?? inicial; }    catch { return inicial; }  });  const set \= useCallback((nouValor: T) \=\> {    setValor(nouValor);    localStorage.setItem(clau, JSON.stringify(nouValor));  }, \[clau\]);  return \[valor, set\] as const;}// HOOK DE FEATURE: cerca de productes (composa els 3 hooks base)function useCercaProductes() {  const \[textCerca, setTextCerca\] \= useState('');  const \[categoria, setCategoria\] \= useLocalStorage('categoria-cerca', 'tots');  // ↑ hook base 3: la categoria persists entre sessions  const cercaRetardada \= useDebounce(textCerca, 400);  // ↑ hook base 2: no fem fetch en cada tecla  const url \= cercaRetardada || categoria \!== 'tots'    ? \`/api/productes?q=${cercaRetardada}\&cat=${categoria}\`    : null;  const { dades: productes, loading, error } \= useFetch\<Producte\[\]\>(url);  // ↑ hook base 1: fetch amb race condition prevention  // Lògica de negoci pròpia del hook de feature:  const resultats \= useMemo(    () \=\> productes?.filter(p \=\> p.estoc \> 0\) ?? \[\],    \[productes\]  );  const netejar \= useCallback(() \=\> {    setTextCerca('');    setCategoria('tots');  }, \[setCategoria\]);  return {    textCerca,    setTextCerca,    categoria,    setCategoria,    resultats,    loading,    error,    netejar,    teResultats: resultats.length \> 0,  };}// Component: completament declaratfunction PàginaCerca() {  const {    textCerca, setTextCerca,    categoria, setCategoria,    resultats, loading, error,    netejar, teResultats,  } \= useCercaProductes();  return (    \<div\>      \<input value={textCerca} onChange={e \=\> setTextCerca(e.target.value)} /\>      \<select value={categoria} onChange={e \=\> setCategoria(e.target.value)}\>        \<option value="tots"\>Tots\</option\>        \<option value="roba"\>Roba\</option\>      \</select\>      \<button onClick={netejar}\>Netejar\</button\>      {loading && \<Spinner /\>}      {error && \<Error missatge={error.message} /\>}      {\!teResultats && \!loading && \<p\>Sense resultats\</p\>}      \<LlistaProductes productes={resultats} /\>    \</div\>  );} |

**Patró 3: Pipeline de transformació de dades**

Cada hook transforma les dades del hook anterior, creant un pipeline de processament.

| // PIPELINE: fetch → normalitzar → filtrar → paginar → ordenar// Pas 1: Obtenir dades brutes de l'APIfunction useProductesRaw() {  return useFetch\<ProducteAPI\[\]\>('/api/productes');}// Pas 2: Normalitzar dades (transformar l'estructura de l'API)function useProductesNormalitzats() {  const { dades: raw, loading, error } \= useProductesRaw();  const productes \= useMemo(    () \=\> raw?.map(p \=\> ({      id: p.product\_id,          // ← normalitzar nom del camp      nom: p.product\_name,      preu: p.price\_eur / 100,  // ← convertir de cèntims a euros      imatge: p.img\_url ?? '/placeholder.jpg', // ← valor per defecte    })) ?? \[\],    \[raw\]  );  return { productes, loading, error };}// Pas 3: Filtrarfunction useProductesFiltrats(filtres: Filtres) {  const { productes, loading, error } \= useProductesNormalitzats();  const filtrats \= useMemo(    () \=\> productes.filter(p \=\>      (\!filtres.categoria || p.categoria \=== filtres.categoria) &&      (\!filtres.preuMax || p.preu \<= filtres.preuMax) &&      (\!filtres.cerca || p.nom.toLowerCase().includes(filtres.cerca))    ),    \[productes, filtres\]  );  return { productes: filtrats, total: filtrats.length, loading, error };}// Pas 4: Paginarfunction useProductesPaginats(filtres: Filtres, perPàgina \= 10\) {  const { productes, total, loading, error } \= useProductesFiltrats(filtres);  const \[pàgina, setPàgina\] \= useState(1);  const productesDeLayPàgina \= useMemo(    () \=\> productes.slice((pàgina \- 1) \* perPàgina, pàgina \* perPàgina),    \[productes, pàgina, perPàgina\]  );  const totalPàgines \= Math.ceil(total / perPàgina);  return {    productes: productesDeLayPàgina,    pàgina,    totalPàgines,    anarAPàgina: setPàgina,    loading,    error,  };}// Component: usa el hook de més alt nivell → sense saber res dels passos intermedisfunction LlistaProductes({ filtres }: { filtres: Filtres }) {  const { productes, pàgina, totalPàgines, anarAPàgina, loading } \=    useProductesPaginats(filtres);  return (/\* JSX paginat \*/);} |

**Patró 4: Composar estat \+ efectes \+ callbacks en un hook de formulari complet**

| // Hook complet per a un formulari de login (compon múltiples preocupacions)function useLoginForm() {  // Composar useState per a cada camp  const \[email, setEmail\] \= useState('');  const \[password, setPassword\] \= useState('');  const \[mostrarPassword, setMostrarPassword\] \= useState(false);  // Composar useLocalStorage per recordar l'email  const \[emailRecordat, setEmailRecordat\] \= useLocalStorage('email-recordat', '');  useEffect(() \=\> {    if (emailRecordat) setEmail(emailRecordat);  }, \[\]);  // Composar useFetch (indirectament via el submit)  const \[loading, setLoading\] \= useState(false);  const \[error, setError\] \= useState\<string | null\>(null);  // Composar useDebounce per a validació en temps real  const emailRetardat \= useDebounce(email, 300);  const emailValid \= emailRetardat.includes('@') && emailRetardat.includes('.');  const handleSubmit \= useCallback(async (e: React.FormEvent, recordar: boolean) \=\> {    e.preventDefault();    setLoading(true);    setError(null);    try {      await authService.login(email, password);      if (recordar) setEmailRecordat(email);    } catch (err) {      setError((err as Error).message);    } finally {      setLoading(false);    }  }, \[email, password, setEmailRecordat\]);  return {    // Estat dels camps    email, setEmail,    password, setPassword,    mostrarPassword, setMostrarPassword,    // Validació    emailValid,    esValid: emailValid && password.length \>= 8,    // Estat async    loading,    error,    // Acció    handleSubmit,  };}// Component net: únicament JSXfunction FormulariLogin() {  const {    email, setEmail,    password, setPassword,    mostrarPassword, setMostrarPassword,    emailValid, esValid,    loading, error,    handleSubmit,  } \= useLoginForm();  return (    \<form onSubmit={(e) \=\> handleSubmit(e, true)}\>      \<input type\="email" value={email} onChange={e \=\> setEmail(e.target.value)}             className={emailValid ? 'valid' : ''} /\>      \<div\>        \<input type\={mostrarPassword ? 'text' : 'password'}               value={password} onChange={e \=\> setPassword(e.target.value)} /\>        \<button type\="button" onClick={() \=\> setMostrarPassword(v \=\> \!v)}\>          {mostrarPassword ? 'Amaga' : 'Mostra'}        \</button\>      \</div\>      {error && \<span\>{error}\</span\>}      \<button type="submit" disabled={\!esValid || loading}\>        {loading ? 'Entrant...' : 'Entrar'}      \</button\>    \</form\>  );} |

**Resum: patrons de composició**

PATRÓ 1: Base → Domini  
  hooks atòmics (useWindowSize, useMediaQuery)  
  → composar en hook de domini (useResponsiveLayout)  
  → el component usa l'abstracció d'alt nivell

PATRÓ 2: Fetch \+ Estat Local \+ Lògica  
  useFetch \+ useDebounce \+ useLocalStorage  
  → composar en useCercaProductes  
  → cada hook resol una responsabilitat

PATRÓ 3: Pipeline de Transformació  
  useProductesRaw → useProductesNormalitzats  
  → useProductesFiltrats → useProductesPaginats  
  → cada pas transforma les dades del pas anterior

PATRÓ 4: Feature Hook Complet  
  useState \+ useLocalStorage \+ useDebounce \+ callbacks  
  → un sol hook d'alt nivell per a la feature  
  → el component és únicament JSX

LA REGLA:  
  Quant de detall veus al component?  
  → Massa detall → extreure un hook  
  → Zero detall (tot al hook) → el hook és massa gran → dividir  
  → Equilibri: el component explica QUÈ fa, el hook explica COM
